一条 SQL 的力量
-- 1. 注册模型（任何 OpenAI 兼容端点均可） SELECT ai.add_completion_model('my-chat', 'https://api.deepseek.com/v1/chat/completions', '{"model": "deepseek-chat", "temperature": 0.3}'::jsonb, 'sk-your-token', 'deepseek'); -- 2. 设默认模型（可选） SET ai.completion_model = 'my-chat'; -- 3. 一条 SQL 完成 AI 任务 SELECT ai.summarize('一段长文本…'); -- 摘要生成 SELECT ai.sentiment('物流太慢了，失望'); -- 情感分类 -- 整列加工：无需导出数据、无需 ETL UPDATE feedback SET summary = ai.summarize(content), is_positive = ai.generate_bool('好评返回true：' || content);
